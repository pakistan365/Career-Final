// ============================================================
// Career Pakistan — api/submit.js
// Vercel Serverless Function — Public content submission
//
// Receives a submission from submit.html, validates it against the
// schema for the chosen sheet, stamps a few server-controlled fields,
// and forwards it to the Google Apps Script Web App, which appends it
// as a new row in the matching Google Sheet tab. Content goes live
// immediately (no moderation queue), per project decision.
//
// Required env var (Vercel → Project → Settings → Environment
// Variables): GOOGLE_SCRIPT_URL — the /exec URL from the Apps Script
// deployment. See apps-script/SETUP-GUIDE.md for how to get this.
// ============================================================

'use strict';

// Mirrors the columns in submit.html / js/google-sheet-loader.js exactly.
const SCHEMAS = {
  Jobs: {
    required: ['Title', 'Organization', 'Apply Link'],
    optional: ['Type', 'Location', 'Salary', 'Deadline', 'Category', 'Province',
      'Experience', 'Education', 'Tags', 'Short Description', 'Image Link',
      'Details', 'PDF Link', 'Source Link'],
  },
  Scholarships: {
    required: ['Title', 'Apply Link'],
    optional: ['Country', 'Funding', 'Deadline', 'Eligibility', 'Type', 'Level',
      'Field', 'University', 'Province', 'Tags', 'Short Description',
      'Image Link', 'Details', 'PDF Link'],
  },
  Internships: {
    required: ['Title', 'Organization', 'Apply Link'],
    optional: ['Location', 'Stipend', 'Duration', 'Deadline', 'Type', 'Category',
      'Tags', 'Short Description', 'Education Level', 'Image Link', 'Details', 'PDF Link'],
  },
  Exams: {
    required: ['Title'],
    optional: ['Exam Type', 'Test Date', 'Registration Deadline', 'Apply Link',
      'Conducting Body', 'Fee', 'Eligibility', 'Syllabus Link', 'Past Papers Link',
      'Tags', 'Province', 'Short Description', 'Image Link', 'Details', 'PDF Link'],
  },
  Books: {
    required: ['Title'],
    optional: ['Author', 'Exam Type', 'Price', 'Apply Link', 'Category', 'Language',
      'Pages', 'Edition', 'Is Free', 'Tags', 'Short Description', 'Download Link',
      'Image Link', 'Details', 'PDF Link'],
  },
  Blogs: {
    required: ['Title', 'Description'],
    optional: ['Category', 'Short Description', 'Image URL', 'Author', 'Date', 'Tags',
      'Apply Link', 'PDF Link', 'Related Jobs Tags', 'Related Exams Tags', 'Read Time'],
  },
  UniversityAdmissions: {
    required: ['University Name', 'Apply Link'],
    optional: ['Short Name', 'Program(s) Offered', 'Degree Level', 'Admission Open Date',
      'Last Date to Apply', 'Entry Test Required', 'Entry Test Name', 'Entry Test Date',
      'Minimum Marks Required', 'Minimum Percentage', 'Eligible Boards', 'Merit List URL',
      'Prospectus Link', 'Fee (Approximate)', 'Location / City', 'Province',
      'University Type', 'Category / Faculty', 'Seats Available', 'Hostel Available',
      'Scholarships Available', 'Image Link', 'Short Description', 'Full Details', 'Tags'],
  },
};

const VALID_SHEETS = Object.keys(SCHEMAS);
const MAX_FIELD_LENGTH = 5000;

function setCors(res) {
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
}

function sanitize(value) {
  return String(value == null ? '' : value).trim().slice(0, MAX_FIELD_LENGTH);
}

async function handler(req, res) {
  setCors(res);
  res.setHeader('Cache-Control', 'no-store');

  if (req.method === 'OPTIONS') return res.status(204).end();
  if (req.method !== 'POST') return res.status(405).json({ ok: false, error: 'Method not allowed' });

  const scriptUrl = process.env.GOOGLE_SCRIPT_URL;
  if (!scriptUrl) {
    return res.status(500).json({
      ok: false,
      error: 'Server is not configured yet: GOOGLE_SCRIPT_URL environment variable is missing.',
    });
  }

  let body = req.body;
  if (typeof body === 'string') {
    try { body = JSON.parse(body); } catch { body = {}; }
  }
  body = body || {};

  // Honeypot: a hidden field real users never touch. If it has a value,
  // pretend success so bots don't learn anything and move on.
  if (sanitize(body.website)) {
    return res.status(200).json({ ok: true, message: 'Thanks! Your submission is now live on the site.' });
  }

  const sheet = sanitize(body.sheet);
  if (!VALID_SHEETS.includes(sheet)) {
    return res.status(400).json({
      ok: false,
      error: `Invalid or missing "sheet". Valid values: ${VALID_SHEETS.join(', ')}`,
    });
  }

  const schema = SCHEMAS[sheet];
  const rawFields = body.fields && typeof body.fields === 'object' ? body.fields : {};

  const fields = {};
  const missing = [];

  for (const key of schema.required) {
    const val = sanitize(rawFields[key]);
    if (!val) missing.push(key);
    fields[key] = val;
  }
  if (missing.length) {
    return res.status(400).json({
      ok: false,
      error: `Missing required field(s): ${missing.join(', ')}`,
    });
  }

  for (const key of schema.optional) {
    fields[key] = sanitize(rawFields[key]);
  }

  // Server-controlled fields — the client never sets these directly.
  fields['Posted Date'] = new Date().toISOString().slice(0, 10); // YYYY-MM-DD
  fields['Is Featured'] = 'No';
  fields['Is Active'] = 'Yes';
  fields['Is Published'] = 'Yes';

  try {
    const upstream = await fetch(scriptUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ sheet, fields }),
      redirect: 'follow',
    });

    const text = await upstream.text();
    let parsed;
    try { parsed = JSON.parse(text); } catch { parsed = null; }

    if (!upstream.ok || !parsed || parsed.ok === false) {
      console.error('[CareerPK] submit.js: Apps Script error:', text);
      return res.status(502).json({
        ok: false,
        error: (parsed && parsed.error) || 'The sheet did not accept this submission. Please try again later.',
      });
    }

    return res.status(200).json({
      ok: true,
      message: 'Thanks! Your submission is now live on the site.',
    });
  } catch (err) {
    console.error('[CareerPK] submit.js: fetch error:', err.message);
    return res.status(500).json({ ok: false, error: 'Failed to reach the submission service.' });
  }
}

module.exports = handler;
